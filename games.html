const BACKEND_URL = "https://bomis-sports-backend.onrender.com"; // 1. Function to update the Live Count sidebar async function updateLiveCounts() { try { const res = await fetch(`${BACKEND_URL}/api/live-counts`); if (!res.ok) throw new Error("Backend not responding"); const counts = await res.json(); const statsHtml = `

Chess: ${counts.chess}/30

Table Tennis: ${counts.tt}/30

Cricket: ${counts.cricket}/30

Football: ${counts.football}/30

`; const container = document.getElementById('stats-container'); if (container) container.innerHTML = statsHtml; } catch (e) { console.error("Dashboard error:", e); const container = document.getElementById('stats-container'); if (container) container.innerHTML = "

Updating counts...

"; } } // 2. Function to submit the selection async function submitSelection() { const indoor = document.querySelector('input[name="indoor"]:checked'); const outdoor = document.querySelector('input[name="outdoor"]:checked'); if (!indoor || !outdoor) { alert("Please select both an Indoor and an Outdoor game!"); return; } const enrollment_no = localStorage.getItem('enrollment_no'); // UI Feedback const mainUi = document.getElementById('main-ui'); const originalContent = mainUi.innerHTML; mainUi.innerHTML = "

Saving your selection... Please wait.

"; try { const res = await fetch(`${BACKEND_URL}/api/save-selection`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ enrollment_no: enrollment_no, indoor_game: indoor.value, outdoor_game: outdoor.value }) }); const result = await res.json(); if (result.success) { localStorage.clear(); // End session mainUi.innerHTML = `

Success! ✅

Your selection has been recorded.

`; updateLiveCounts(); // Refresh the sidebar } else { alert("Error: " + (result.message || "Could not save.")); mainUi.innerHTML = originalContent; } } catch (e) { alert("Connection failed. Check if backend is live."); mainUi.innerHTML = originalContent; } } // Run on page load updateLiveCounts();
