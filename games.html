const BACKEND_URL = "https://bomis-sports-backend.onrender.com"; // Updates the dashboard UI by fetching data from the backend async function updateLiveCounts() { try { const res = await fetch(`${BACKEND_URL}/api/live-counts`); const counts = await res.json(); const statsHtml = `

Chess: ${counts.chess}/30

Table Tennis: ${counts.tt}/30

Cricket: ${counts.cricket}/30

Football: ${counts.football}/30

`; document.getElementById('stats-container').innerHTML = statsHtml; } catch (e) { console.error("Dashboard error:", e); } } // Submits selection to backend async function submitSelection() { const indoor = document.querySelector('input[name="indoor"]:checked'); const outdoor = document.querySelector('input[name="outdoor"]:checked'); if(!indoor || !outdoor) return alert("Select both games!"); try { const res = await fetch(`${BACKEND_URL}/api/save-selection`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ enrollment_no: localStorage.getItem('enrollment_no'), indoor_game: indoor.value, outdoor_game: outdoor.value }) }); const result = await res.json(); if(result.success) { alert("Selection Saved Successfully!"); window.location.href = "https://bomis-lbnagar.com/"; } } catch (e) { alert("Connection failed."); } } updateLiveCounts();
