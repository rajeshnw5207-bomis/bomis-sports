const BACKEND_URL = "https://bomis-sports-backend.onrender.com"; // 1. Fetch and Display Live Counts async function updateLiveCounts() { try { const res = await fetch(`${BACKEND_URL}/api/live-counts`); if (!res.ok) throw new Error("Server response error"); const counts = await res.json(); // Ensure keys match the backend JSON exactly const statsHtml = `

Chess: ${counts.chess || 0}/30

Table Tennis: ${counts.tt || 0}/30

Cricket: ${counts.cricket || 0}/30

Football: ${counts.football || 0}/30

`; document.getElementById('stats-container').innerHTML = statsHtml; } catch (e) { console.error("Dashboard error:", e); document.getElementById('stats-container').innerHTML = "

Backend waking up... Please wait.

"; // Retry once after 5 seconds if it fails setTimeout(updateLiveCounts, 5000); } } // 2. Submit Logic async function submitSelection() { const indoor = document.querySelector('input[name="indoor"]:checked'); const outdoor = document.querySelector('input[name="outdoor"]:checked'); const enrollment = localStorage.getItem('enrollment_no'); if (!indoor || !outdoor) { alert("Please select both games!"); return; } const mainUi = document.getElementById('main-ui'); const originalContent = mainUi.innerHTML; mainUi.innerHTML = "

Saving your selection... Please wait.

"; try { const res = await fetch(`${BACKEND_URL}/api/save-selection`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ enrollment_no: enrollment, indoor_game: indoor.value, outdoor_game: outdoor.value }) }); const result = await res.json(); if (result.success) { localStorage.clear(); // Clear session on success mainUi.innerHTML = `

Success! ✅

Your selection has been recorded.

`; updateLiveCounts(); } else { alert("Error: " + (result.message || "Could not save.")); mainUi.innerHTML = originalContent; } } catch (e) { alert("Connection failed. Check your internet or backend status."); mainUi.innerHTML = originalContent; } } // Initialize updateLiveCounts();
